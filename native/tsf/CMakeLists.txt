cmake_minimum_required(VERSION 3.20)
project(CantoneseIME LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "CantoneseIME is a Windows TSF in-process COM DLL")
endif()
add_library(CantoneseIME SHARED tsf.cpp ipc.cpp registration.cpp tsf_guids.cpp CantoneseIME.def CantoneseIME.rc)
target_compile_features(CantoneseIME PRIVATE cxx_std_17)
target_compile_definitions(CantoneseIME PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_link_libraries(CantoneseIME PRIVATE ole32 oleaut32 uuid user32 gdi32 advapi32 shell32)
set_target_properties(CantoneseIME PROPERTIES PREFIX "" OUTPUT_NAME "CantoneseIME")
if(MSVC)
  target_compile_options(CantoneseIME PRIVATE /W4 /EHsc /utf-8)
else()
  target_compile_options(CantoneseIME PRIVATE -Wall -Wextra)
  target_link_options(CantoneseIME PRIVATE -static -Wl,--no-undefined -Wl,--dynamicbase -Wl,--nxcompat)
endif()
